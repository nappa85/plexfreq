cmake_minimum_required(VERSION 3.16)
project(PlexFreq VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Multimedia DBus)
find_program(CARGO cargo REQUIRED)
set(RUST_LIBRARY "${CMAKE_SOURCE_DIR}/target/release/libplexfreq_core.a")
file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/*.rs")
add_custom_command(OUTPUT "${RUST_LIBRARY}"
    COMMAND "${CARGO}" build --locked --release --lib
    WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    DEPENDS ${RUST_SOURCES} Cargo.toml Cargo.lock)
add_custom_target(rust_core DEPENDS "${RUST_LIBRARY}")
add_executable(plexfreq app/src/main.cpp app/src/backend.cpp app/src/backend.h app/src/mpris.cpp app/src/mpris.h app/desktop.qrc)
add_dependencies(plexfreq rust_core)
target_link_libraries(plexfreq PRIVATE "${RUST_LIBRARY}" Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::DBus pthread dl m)
include(GNUInstallDirs)
install(TARGETS plexfreq RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/plexfreq.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES app/icons/harbour-plexfreq.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME plexfreq.svg)
include(CTest)
if(BUILD_TESTING)
find_package(Qt6 REQUIRED COMPONENTS Test Network)
add_executable(qt_backend_test tests/qt_backend.cpp tests/qt-backend-test.qrc app/src/backend.cpp app/src/backend.h app/src/mpris.cpp app/src/mpris.h)
add_dependencies(qt_backend_test rust_core)
target_include_directories(qt_backend_test PRIVATE app/src)
target_link_libraries(qt_backend_test PRIVATE "${RUST_LIBRARY}" Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Multimedia Qt6::Test Qt6::Network Qt6::DBus pthread dl m)
find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
add_test(NAME native_backend_integration COMMAND "${DBUS_RUN_SESSION}" -- $<TARGET_FILE:qt_backend_test>)
set_tests_properties(native_backend_integration PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 40)
add_test(NAME desktop_qml_smoke COMMAND bash "${CMAKE_SOURCE_DIR}/tools/smoke-test.sh" $<TARGET_FILE:plexfreq>)
set_tests_properties(desktop_qml_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 20)
endif()
